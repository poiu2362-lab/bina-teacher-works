async function uploadPrivate(file) { if (!file) { throw new Error('لم يتم اختيار ملف'); } if (file.size > 20 * 1024 * 1024) { throw new Error('الحد الأعلى للمرفق 20 ميجابايت'); } const user = auth().currentUser; if (!user) { throw new Error('يجب تسجيل الدخول أولاً'); } const idToken = await user.getIdToken(true); // الحصول على رابط الرفع الآمن const signResponse = await fetch('/api/upload', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${idToken}` }, body: JSON.stringify({ fileName: file.name, contentType: file.type || 'application/octet-stream', fileSize: file.size }) }); let signData = {}; try { signData = await signResponse.json(); } catch (_) {} if (!signResponse.ok) { throw new Error( signData.error || 'تعذر تجهيز رابط رفع المرفق' ); } if (!signData.uploadUrl) { throw new Error('لم يتم إنشاء رابط رفع المرفق'); } // رفع الملف مباشرة إلى Vercel Blob const uploadResponse = await fetch(signData.uploadUrl, { method: 'PUT', body: file }); if (!uploadResponse.ok) { let details = ''; try { details = await uploadResponse.text(); } catch (_) {} console.error( 'Blob upload failed:', uploadResponse.status, details ); throw new Error( `تعذر رفع المرفق (${uploadResponse.status})` ); } return { pathname: signData.pathname, fileName: file.name, fileType: file.type || 'application/octet-stream', fileSize: file.size }; }
